# -- OAI-AdsBot: WAJIB untuk peninjauan ChatGPT Ads -------------------------
# Memvalidasi keamanan & relevansi halaman yang didaftarkan sebagai iklan.
# Datanya TIDAK dipakai melatih model. Tanpa akses bot ini, peninjauan akun
# dan iklan gagal ("situs web bisnis tidak tersedia").
User-agent: OAI-AdsBot
Allow: /

# -- OAI-SearchBot: disarankan OpenAI ---------------------------------------
# Indeks pencarian ChatGPT; membantu OpenAI memahami konten publik situs.
User-agent: OAI-SearchBot
Allow: /

# -- ChatGPT-User: kunjungan atas permintaan pengguna ------------------------
# Bukan perayapan otomatis — dipicu saat pengguna ChatGPT membuka tautan ini.
User-agent: ChatGPT-User
Allow: /

# -- Semua perayap lain ------------------------------------------------------
# CATATAN: aturan wildcard ini juga mengizinkan GPTBot, yaitu perayap yang
# mengumpulkan konten untuk PELATIHAN model — dan itu TIDAK dibutuhkan untuk
# beriklan. Kalau konten situs ini tidak ingin dipakai melatih model, tambahkan
# blok berikut di atas baris User-agent: * —
#
#   User-agent: GPTBot
#   Disallow: /
#
# Menghapus blok GPTBot saja tidak cukup; wildcard di bawah tetap mengizinkan.
User-agent: *
